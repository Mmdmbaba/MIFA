    42→interface WizardState {
    43→  step: number
    44→  name: string
    45→  income: string
    46→  payday: number

    57→
    58→function initialState(data?: ProfileData): WizardState {
    59→  const p = data?.profile
    60→  return {
    61→    step: 0,
    62→    name: p?.name ?? "",
    63→    income: p ? String(p.income) : "",
    64→    payday: p?.payday ?? 1,
    65→    cycleStartDay: p?.cycleStartDay ?? 1,
    66→    calMonth: currentMonthKey(),
    67→    calOpen: false,
    68→    fixed: (data?.fixed ?? []).map((f) => ({ label: f.label, amount: String(f.amount), icon: f.icon })),
    69→    customLabel: "",
    70→    customAmount: "",
    71→    savingsMode: (p?.savingsMode as SavingsMode) ?? "auto",
    72→    savingsAmount: p?.savingsMode === "amount" ? String(p.savingsAmount) : "",
    73→    savingsPercent: p?.savingsPercent ?? 20,
    74→  }
    75→}
